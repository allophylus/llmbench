  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;

        const f = (1 - TANG) * 0.5;
        if (!a.settled) { a.vx += tvx * f; a.vy += tvy * f; a.vz += tvz * f; }
        if (!b.settled) { b.vx -= tvx * f; b.vy -= tvy * f; b.vz -= tvz * f; }
      }
    }

    else if (b.z > ZMAX) { b.z = ZMAX; b.vz = -b.vz * REST; }
