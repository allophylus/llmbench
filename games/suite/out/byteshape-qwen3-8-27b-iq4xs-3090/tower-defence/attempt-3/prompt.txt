    S.breakLeft--;
    if(S.breakLeft<=0) startWave();
  } else {
    // spawn
    if(S.spawned<S.spawnLeft){
      S.spawnAcc+=dt;
      while(S.spawnAcc>=SPAWN_GAP/60 && S.spawned<S.spawnLeft){ S.spawnAcc-=SPAWN_GAP/60; spawnUnit(); }
    }
    // move
    for(const u of S.units.slice()){
      u.wp+=SPEED*dt;
      if(u.wp>=PATH.length-1){ S.lives--; removeUnit(u); if(S.lives<=0){S.lives=0;S.over=true;S.won=false;showOverlay(false);} continue; }
      const p=unitPos(u.wp); u.x=p.x; u.y=p.y;
      u.mesh.position.copy(unitWorld(u.wp));
      const key=u.x+','+u.y;
      if(S.unitAt[key]!==u){ delete S.unitAt[S.unitAt[key].x+','+S.unitAt[key].y]; delete S.hpAt[S.unitAt[key].x+','+S.unitAt[key].y]; }
      S.unitAt[key]=u; S.hpAt[key]=u.hp;
    }
    combat();
    if(S.spawned>=S.spawnLeft && S.units.length===0) endWave();
  }
  // tracers
  for(const t of S.tracers.slice()){ t.life-=dt; if(t.life<=0){fxGroup.remove(t.line);S.tracers.splice(S.tracers.indexOf(t),1);} }
  updateHUD();
}
