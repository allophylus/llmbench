  }
  return null;
}

/* ---------- ambient background ---------- */
const bg = document.getElementById('bg');
const bctx = bg.getContext('2d');
let BW = 0, BH = 0, motes = [];
function sizeBG(){
  BW = bg.width = innerWidth; BH = bg.height = innerHeight;
  motes = [];
  const n = Math.min(70, Math.floor(BW * BH / 26000));
  for (let i = 0; i < n; i++) motes.push({
    x: Math.random() * BW, y: Math.random() * BH,
    r: 0.6 + Math.random() * 2.2, s: 4 + Math.random() * 14,
    a: 0.05 + Math.random() * 0.22, ph: Math.random() * 6.28
  });
}
function drawBG(t){
  const g = bctx.createLinearGradient(0, 0, 0, BH);
  g.addColorStop(0, '#07131a'); g.addColorStop(0.55, '#0c2230'); g.addColorStop(1, '#0a1c26');
  bctx.fillStyle = g; bctx.fillRect(0, 0, BW, BH);
  const rg = bctx.createRadialGradient(BW * 0.5, BH * 0.42, 0, BW * 0.5, BH * 0.42, Math.max(BW, BH) * 0.7);
  rg.addColorStop(0, 'rgba(55,224,200,0.10)'); rg.addColorStop(1, 'rgba(55,224,200,0)');
  bctx.fillStyle = rg; bctx.fillRect(0, 0, BW, BH);
  for (const m of motes){
    m.y -= m.s * 0.016; if (m.y < -4){ m.y = BH + 4; m.x = Math.random() * BW; }
    const tw = 0.6 + 0.4 * Math.sin(t * 0.001 + m.ph);
    bctx.globalAlpha = m.a * tw;
    bctx.fillStyle = '#8fe6d8';
    bctx.beginPath(); bctx.arc(m.x, m.y, m.r, 0, 6.283); bctx.fill();
  }
  bctx.globalAlpha = 1;
}

/* ---------- animation loop ---------- */
const clock = new THREE.Clock();
function animate(){
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = performance.now();
  drawBG(t);
  for (let r = 0; r < N; r++) for (let c = 0; c < N; c++){
    const g = gems[r][c]; if (!g) continue;
    const u = g.userData;
    const tx = (c - (N - 1) / 2) * SP, ty = ((N - 1) / 2 - r) * SP;
    g.position.x += (tx - g.position.x) * Math.min(1, dt * 14);
    g.position.y += (ty - g.position.y) * Math.min(1, dt * 14);
    if (u.spawn > 0){
      u.spawn -= dt * 3.2;
      const s = u.spawn > 0 ? Math.max(0.001, u.spawn) : 1;
      g.scale.setScalar(s * u.targetScale);
      if (u.spawn <= 0) u.spawn = 0;
    } else {
      g.scale.setScalar(u.targetScale);
    }
    g.rotation.y += dt * u.spin;
    g.rotation.x += dt * u.spin * u.spinAxis;
  }
  boardGroup.rotation.y = Math.sin(t * 0.00022) * 0.05;
  boardGroup.rotation.x = Math.sin(t * 0.00017) * 0.03;
  if (shownScore < score){
    shownScore += (score - shownScore) * Math.min(1, dt * 8);
    if (score - shownScore < 0.5) shownScore = score;
    vScore.textContent = Math.round(shownScore).toLocaleString();
  }
  renderer.render(scene, camera);
}
